   14  -- decision_value_pinned proves the decision CHECK constraint's vocabulary is
   15  -- the EXACT closed three-value set {accepted_internal_with_limitation,
   16  -- accepted_funder_with_limitation, accepted_public_with_limitation} and no
   17  -- other value - not merely that those three substrings appear somewhere in

   19  -- substring check would silently keep passing even if a third value were
   20  -- ever added to the constraint). It parses the actual
   21  -- pg_get_constraintdef() `= ANY (ARRAY[...])` expression Postgres normalizes
   22  -- an `IN (...)` CHECK into, extracts every literal in that array, and
   23  -- compares the resulting set against the expected two-element set for exact
   24  -- equality in both directions (every expected value present, no unexpected
   25  -- value present).
   26  WITH decision_constraint AS (
   27    SELECT c.oid, pg_get_constraintdef(c.oid) AS def
   28      FROM pg_constraint c
   29      JOIN pg_class r ON r.oid = c.conrelid

   33       AND c.conname = 'coverage_review_decisions_p2_10_decision_check'
   34  ),

   36    SELECT regexp_replace(def, '^.*=\s*ANY\s*\(ARRAY\[(.*)\]\)\).*$', '\1') AS array_body
